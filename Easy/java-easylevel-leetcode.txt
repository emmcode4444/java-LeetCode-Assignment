---------Java LeetCode 1st problem Solution : TWO SUM------>
-----------------------------------------------------------

class Solution {
    public int[] twoSum(int[] nums, int target) {
        for(int i = 0; i<nums.length; i++){
            for(int j = i+1; j<nums.length; j++){
                if(nums[i] + nums[j] == target){
                    int a[] = {i,j};
                    return a;
                }
            }
        }
        return null;
    }
}

---------Java LeetCode 9th problem Solution : Palindrome Number------>
---------------------------------------------------------------------

class Solution {
    public boolean isPalindrome(int x) {
        if(x<0){
            return false;
        }

        int n =x;
        int reverseNumber = 0;

        while(n>0){
            int lastDigit= n%10;
            reverseNumber = reverseNumber*10+lastDigit;

            n = n/10;

        }
        if(reverseNumber == x){
            return true;
        }
        else{
            return false;
        }

    }
}
