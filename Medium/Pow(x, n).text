class Solution {
    public double myPow(double x, int n) {
        long N = n;
          if(n < 0){
            N = -N;
          }
          double result = 1;
          while(N!=0){
            if(N%2 !=0){
                result = result * x;
            }
            x = x*x;
            N = N/2;
          }
          if(n < 0){
            result = 1/result;
          }
          return result;
    }
}
